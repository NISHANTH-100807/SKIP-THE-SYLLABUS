// Binary Search Tree (Java OOP) public class BSTNode { int val; BSTNode left, right; public BSTNode(int v) { this.val = v; } } public class BST { private BSTNode root; public void insert(int val) { root = insertRec(root, val); } private BSTNode insertRec(BSTNode node, int val) { if (node == null) return new BSTNode(val); if (val < node.val) node.left = insertRec(node.left, val); else if (val > node.val) node.right = insertRec(node.right, val); return node; // Duplicate keys ignored } }
